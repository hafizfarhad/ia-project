  \begin{scope}[shift={(#2)}]
    \fill[uiS!85] (0,0.26) circle (0.11);
    \fill[uiS!85] (-0.19,-0.02) .. controls (-0.19,0.14) and (0.19,0.14) .. (0.19,-0.02) -- cycle;
    \node[sub, anchor=north] (#1) at (0,-0.04) {#3};
  \end{scope}}
